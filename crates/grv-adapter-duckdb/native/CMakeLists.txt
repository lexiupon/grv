cmake_minimum_required(VERSION 3.24)
project(grv_duckdb_native CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_UNITTESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHELL OFF CACHE BOOL "" FORCE)
set(DISABLE_BUILTIN_EXTENSIONS ON CACHE BOOL "" FORCE)
set(BUILD_COMPLETE_EXTENSION_SET OFF CACHE BOOL "" FORCE)
set(OVERRIDE_GIT_DESCRIBE "v1.5.6" CACHE STRING "" FORCE)
set(GIT_COMMIT_HASH "069cc9f" CACHE STRING "" FORCE)
add_subdirectory(${GRV_DUCKDB_SOURCE} duckdb)
if(APPLE)
  set(GRV_NATIVE_LIBRARIES duckdb_static duckdb_generated_extension_loader core_functions_extension parquet_extension)
else()
  set(GRV_NATIVE_LIBRARIES "$<LINK_GROUP:RESCAN,duckdb_static,duckdb_generated_extension_loader,core_functions_extension,parquet_extension>")
endif()
add_library(grv_duckdb_guard SHARED shim.cpp)
target_include_directories(grv_duckdb_guard PRIVATE ${GRV_DUCKDB_SOURCE}/src/include ${GRV_DUCKDB_SOURCE}/extension/parquet/include ${GRV_DUCKDB_SOURCE}/third_party/mbedtls/include)
target_link_libraries(grv_duckdb_guard PRIVATE ${GRV_NATIVE_LIBRARIES})
add_executable(grv_native_proof proof.cpp)
target_include_directories(grv_native_proof PRIVATE ${GRV_DUCKDB_SOURCE}/src/include)
target_link_libraries(grv_native_proof PRIVATE ${GRV_NATIVE_LIBRARIES})
add_executable(grv_native_fixture fixture.cpp)
target_include_directories(grv_native_fixture PRIVATE ${GRV_DUCKDB_SOURCE}/src/include)
target_link_libraries(grv_native_fixture PRIVATE ${GRV_NATIVE_LIBRARIES})
enable_testing()
add_executable(grv_s3_limits_proof s3_limits_proof.cpp)
add_test(NAME duckdb_verified_s3_metadata_budget COMMAND grv_s3_limits_proof)
add_test(NAME duckdb_initial_bind_callbacks_guarded COMMAND grv_native_proof initial)
add_test(NAME duckdb_nested_view_macro_binding_guarded COMMAND grv_native_proof nested)
add_test(NAME duckdb_metadata_and_replacement_callbacks_guarded COMMAND grv_native_proof metadata)
add_test(NAME duckdb_automatic_rebind_callbacks_guarded COMMAND grv_native_proof rebind)
add_test(NAME duckdb_direct_binding_and_callback_identity_guarded COMMAND grv_native_proof direct)
add_test(NAME duckdb_explicit_implicit_cast_callbacks_guarded COMMAND grv_native_proof cast)
add_test(NAME duckdb_type_modifier_callbacks_guarded COMMAND grv_native_proof type)
